-- Utility
function cleanConnection(con)
    if con then
        pcall(function() con:Disconnect() end)
    end
end

function clearConnections(list)
    for _, con in ipairs(list) do
        cleanConnection(con)
    end
    table.clear(list)
end

function isInWorkspace(obj)
    return obj and obj:IsDescendantOf(Workspace)
end

function getCharacter()
    return LocalPlayer.Character
end

function getHumanoid()
    local char = getCharacter()
    if char then
        return char:FindFirstChildOfClass("Humanoid")
    end
    return nil
end

function getTopModel(part)
    local current = part
    local topModel = nil
    while current and current ~= Workspace do
        if current:IsA("Model") then
            topModel = current
        end
        current = current.Parent
    end
    return topModel
end

function getModelPrimaryPart(model)
    if model.PrimaryPart then
        return model.PrimaryPart
    end
    local best = nil
    local bestVol = -1
    for _, desc in ipairs(model:GetDescendants()) do
        if desc:IsA("BasePart") then
            local vol = desc.Size.X * desc.Size.Y * desc.Size.Z
            if vol > bestVol then
                bestVol = vol
                best = desc
            end
        end
    end
    return best
end

function getCartForward(cartInfo)
    local primary = cartInfo.PrimaryPart
    if primary then
        return primary.CFrame.LookVector
    end
    return cartInfo.Model:GetPivot().LookVector
end

function playTween(object, properties, duration, easingStyle, easingDirection)
    local tween = TweenService:Create(object, TweenInfo.new(duration or 0.15, easingStyle or Enum.EasingStyle.Quad, easingDirection or Enum.EasingDirection.Out), properties)
    tween:Play()
    return tween
end

function styleButton(btn, accent)
    accent = accent or CONFIG.AccentColor
    btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    btn.BorderSizePixel = 0
    btn.TextColor3 = Color3.fromRGB(240, 240, 240)
    btn.Font = Enum.Font.GothamSemibold
    btn.TextSize = 12
    btn.AutoButtonColor = false

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = btn

    local stroke = Instance.new("UIStroke")
    stroke.Color = accent
    stroke.Thickness = 1
    stroke.Transparency = 0.7
    stroke.Parent = btn

    local originalColor = btn.BackgroundColor3
    local hoverColor = Color3.fromRGB(55, 55, 55)

    btn.MouseEnter:Connect(function()
        playTween(btn, {BackgroundColor3 = hoverColor}, 0.12)
        playTween(stroke, {Transparency = 0.3}, 0.12)
    end)
    btn.MouseLeave:Connect(function()
        playTween(btn, {BackgroundColor3 = originalColor}, 0.12)
        playTween(stroke, {Transparency = 0.7}, 0.12)
    end)
    btn.MouseButton1Down:Connect(function()
        playTween(btn, {Size = UDim2.new(btn.Size.X.Scale, btn.Size.X.Offset * 0.95, btn.Size.Y.Scale, btn.Size.Y.Offset * 0.95)}, 0.08)
    end)
    btn.MouseButton1Up:Connect(function()
        playTween(btn, {Size = UDim2.new(btn.Size.X.Scale, btn.Size.X.Offset / 0.95, btn.Size.Y.Scale, btn.Size.Y.Offset / 0.95)}, 0.08)
    end)
end

-- Registry & detection
function getButtonName(part)
    local name = string.lower(part.Name)
    if name == "up" then
        return "Up"
    elseif name == "down" then
        return "Down"
    elseif name == "start" or name == "on" then
        return "Toggle"
    end
    return nil
end

function findClickDetector(part)
    local cd = part:FindFirstChildOfClass("ClickDetector") or part:FindFirstChildWhichIsA("ClickDetector", true)
    if cd then return cd end

    local parent = part.Parent
    if parent then
        cd = parent:FindFirstChildOfClass("ClickDetector")
        if cd then return cd end
        for _, sibling in ipairs(parent:GetChildren()) do
            if sibling:IsA("ClickDetector") then
                return sibling
            end
        end
    end

    return nil
end

function createCartInfo(model)
    return {
        Model = model,
        Buttons = {},
        Seats = {},
        PrimaryPart = nil,
        Esp = {},
        ButtonColors = {},
        ButtonConnections = {},
        Guard = {},
        LastCounter = {},
        LastPosition = nil,
        IsMoving = false,
        MoveStateTimer = 0,
        LockActive = false,
        LockedToggleRed = nil,
        ToggleArmed = true,
        ViewerClone = nil,
    }
end

function scanModelForParts(cartInfo)
    local model = cartInfo.Model
    cartInfo.Buttons = {}
    cartInfo.Seats = {}

    for _, desc in ipairs(model:GetDescendants()) do
        if desc:IsA("BasePart") then
            local role = getButtonName(desc)
            if role then
                local cd = findClickDetector(desc)
                if cd then
                    if role == "Toggle" then
                        if not cartInfo.Buttons.Toggle or string.lower(desc.Name) == "start" then
                            cartInfo.Buttons.Toggle = cd
                        end
                    else
                        cartInfo.Buttons[role] = cd
                    end
                    cartInfo.ButtonColors[cd] = desc.Color
                end
            end
            if desc:IsA("Seat") or desc:IsA("VehicleSeat") then
                table.insert(cartInfo.Seats, desc)
            end
        end
    end

    cartInfo.PrimaryPart = getModelPrimaryPart(model)
end

function registerCart(model)
    if Registry[model] then
        return Registry[model]
    end
    local info = createCartInfo(model)
    Registry[model] = info
    scanModelForParts(info)
    if AllCartsMode and AllCartsLockActive and not info.LockActive then
        toggleLock(info, true)
    end
    return info
end

function unregisterCart(model)
    local info = Registry[model]
    if not info then return end

    if FocusedCart == info then
        unfocusCart()
    end

    clearConnections(info.ButtonConnections)

    if info.Esp.Highlight then info.Esp.Highlight:Destroy() end
    if info.Esp.Nameplate then info.Esp.Nameplate:Destroy() end

    Registry[model] = nil
end

function scanWorkspace()
    for _, desc in ipairs(Workspace:GetDescendants()) do
        if desc:IsA("BasePart") then
            local role = getButtonName(desc)
            if role and findClickDetector(desc) then
                local model = getTopModel(desc)
                if model then
                    registerCart(model)
                end
            end
        end
    end

    for model, info in pairs(Registry) do
        if not isInWorkspace(model) then
            unregisterCart(model)
        end
    end
end

-- Cart state helpers
function getCartRiderInfo(cartInfo)
    for _, seat in ipairs(cartInfo.Seats) do
        local occupant = seat.Occupant
        if occupant then
            local char = occupant.Parent
            local player = Players:GetPlayerFromCharacter(char)
            return true, player and player.Name or char.Name
        end
    end
    return false, nil
end

-- Viewer clone
function buildViewerClone(cartInfo)
    if cartInfo.ViewerClone then
        cartInfo.ViewerClone:Destroy()
    end

    local clone = cartInfo.Model:Clone()
    if not clone then return end
    clone.Name = "CartHubViewerClone"

    for _, desc in ipairs(clone:GetDescendants()) do
        if desc:IsA("Script") or desc:IsA("LocalScript") or desc:IsA("ModuleScript") or desc:IsA("ClickDetector") then
            desc:Destroy()
        elseif desc:IsA("BasePart") then
            desc.Anchored = true
            desc.CanCollide = false
            desc.CanQuery = false
            desc.CanTouch = false
        end
    end

    local cf, size = clone:GetBoundingBox()
    local center = cf.Position
    for _, part in ipairs(clone:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CFrame = part.CFrame - center
        end
    end

    clone.Parent = Gui.ViewerFrame
    cartInfo.ViewerClone = clone
end

-- Controls
function isRedColor(color)
    local r, g, b = color.R * 255, color.G * 255, color.B * 255
    -- broad red detection: red dominant and green/blue low
    return r > 150 and g < 120 and b < 120 and r > g + 30 and r > b + 30
end

function getCartSpeed(cartInfo)
    local model = cartInfo.Model
    local candidates = { "Speed", "speed", "CurrentSpeed", "currentSpeed", "CartSpeed" }
    for _, name in ipairs(candidates) do
        local val = model:FindFirstChild(name)
        if val and (val:IsA("IntValue") or val:IsA("NumberValue")) then
            return val.Value
        end
    end
    for _, child in ipairs(model:GetDescendants()) do
        if table.find(candidates, child.Name) and (child:IsA("IntValue") or child:IsA("NumberValue")) then
            return child.Value
        end
    end
    return nil
end

function fireButton(cd)
    if not cd then return end

    if typeof(fireclickdetector) == "function" then
        pcall(function()
            fireclickdetector(cd, 0, "MouseClick")
        end)
        return
    end

    pcall(function()
        local original = cd.MaxActivationDistance
        cd.MaxActivationDistance = 10000
        cd:Fire(LocalPlayer)
        task.delay(0.05, function()
            pcall(function() cd.MaxActivationDistance = original end)
        end)
    end)
end

function guiPressButton(cartInfo, role)
    local cd = cartInfo.Buttons[role]
    if not cd then return end
    cartInfo.Guard[cd] = tick() + CONFIG.LockGuardDuration
    fireButton(cd)
end

function updateLockButtonVisual()
    local active = false
    if AllCartsMode then
        active = AllCartsLockActive
    elseif FocusedCart then
        active = FocusedCart.LockActive
    end

    local targetColor = active and CONFIG.LockActiveColor or Color3.fromRGB(40, 40, 40)
    local strokeColor = active and CONFIG.LockActiveColor or CONFIG.AccentColor
    local stroke = Gui.LockBtn:FindFirstChildOfClass("UIStroke")
    Gui.LockBtn.BackgroundColor3 = targetColor
    if stroke then
        stroke.Color = strokeColor
    end
end

function toggleLock(cartInfo, forceState)
    local newState = forceState ~= nil and forceState or not cartInfo.LockActive
    cartInfo.LockActive = newState

    if newState then
        clearConnections(cartInfo.ButtonConnections)

        -- Remember target state to defend
        cartInfo.LockedSpeed = getCartSpeed(cartInfo)

        local toggleCd = cartInfo.Buttons.Toggle
        if toggleCd then
            local togglePart = toggleCd.Parent
            if togglePart and togglePart:IsA("BasePart") then
                cartInfo.LockedToggleRed = isRedColor(togglePart.Color)
                cartInfo.ToggleArmed = true
            end
        end

        -- Only use color-based counteract when no Speed value is available
        if cartInfo.LockedSpeed == nil then
            for role, cd in pairs(cartInfo.Buttons) do
                if role == "Toggle" then continue end
                local part = cd.Parent
                if part and part:IsA("BasePart") then
                    cartInfo.ButtonColors[cd] = part.Color

                    local function counteract()
                        if not cartInfo.LockActive then return end
                        if tick() < (cartInfo.Guard[cd] or 0) then return end
                        if tick() < (cartInfo.LastCounter[role] or 0) then return end

                        cartInfo.LastCounter[role] = tick() + CONFIG.LockCounterDebounce
                        if role == "Up" then
                            guiPressButton(cartInfo, "Down")
                        elseif role == "Down" then
                            guiPressButton(cartInfo, "Up")
                        end
                    end

                    table.insert(cartInfo.ButtonConnections, part:GetPropertyChangedSignal("Color"):Connect(function()
                        if not cartInfo.LockActive then return end
                        if tick() < (cartInfo.Guard[cd] or 0) then return end
                        if tick() < (cartInfo.LastCounter[role] or 0) then return end

                        local delta = (part.Color - (cartInfo.ButtonColors[cd] or part.Color)).Magnitude
                        if delta > 0.001 then
                            counteract()
                        end
                    end))

                    table.insert(cartInfo.ButtonConnections, part:GetPropertyChangedSignal("BrickColor"):Connect(function()
                        if not cartInfo.LockActive then return end
                        if tick() < (cartInfo.Guard[cd] or 0) then return end
                        if tick() < (cartInfo.LastCounter[role] or 0) then return end
                        counteract()
                    end))
                end
            end
        end

        for role, _ in pairs(cartInfo.Buttons) do
            cartInfo.LastCounter[role] = tick() + CONFIG.LockCounterDebounce
        end
    else
        clearConnections(cartInfo.ButtonConnections)
        cartInfo.LockedToggleRed = nil
        cartInfo.ToggleArmed = true
        cartInfo.LockedSpeed = nil
    end
end

-- Focus / spectate
function focusCart(cartInfo)
    if FocusedCart == cartInfo and not AllCartsMode then return end
    if not isInWorkspace(cartInfo.Model) then return end

    stopAllSpam()
    unfocusCart()

    AllCartsMode = false
    FocusedCart = cartInfo
    Gui.FocusName.Text = "Spectating: " .. cartInfo.Model.Name
    Gui.ControlPanel.Visible = true
    Gui.ViewerFrame.Visible = true

    local primary = cartInfo.PrimaryPart
    if primary then
        Camera.CameraType = Enum.CameraType.Custom
        Camera.CameraSubject = primary
    end

    buildViewerClone(cartInfo)
    refreshCartList()
end

function focusAllCarts()
    if AllCartsMode then return end

    stopAllSpam()
    unfocusCart()

    AllCartsMode = true
    FocusedCart = nil
    Gui.FocusName.Text = "Controlling: ALL CARTS"
    Gui.ControlPanel.Visible = true
    Gui.ViewerFrame.Visible = false

    local humanoid = getHumanoid()
    if humanoid then
        Camera.CameraSubject = humanoid
    end
    Camera.CameraType = Enum.CameraType.Custom

    if AllCartsLockActive then
        for _, info in pairs(Registry) do
            if not info.LockActive then
                toggleLock(info, true)
            end
        end
    end

    updateLockButtonVisual()
    refreshCartList()
end

function unfocusCart()
    if not FocusedCart and not AllCartsMode then return end

    stopAllSpam()

    local humanoid = getHumanoid()
    if humanoid then
        Camera.CameraSubject = humanoid
    end
    Camera.CameraType = Enum.CameraType.Custom

    if FocusedCart and FocusedCart.ViewerClone then
        FocusedCart.ViewerClone:Destroy()
        FocusedCart.ViewerClone = nil
    end

    if FocusedCart and FocusedCart.LockActive then
        toggleLock(FocusedCart, false)
    end

    if AllCartsMode and AllCartsLockActive then
        for _, info in pairs(Registry) do
            if info.LockActive then
                toggleLock(info, false)
            end
        end
    end

    FocusedCart = nil
    AllCartsMode = false
    Gui.FocusName.Text = "No cart focused"
    Gui.ControlPanel.Visible = false
    Gui.ViewerFrame.Visible = true

    updateLockButtonVisual()
    refreshCartList()
end

function checkFocusedCart()
    if not FocusedCart then return end
    local info = FocusedCart
    if not isInWorkspace(info.Model) then
        unfocusCart()
        return
    end
    local primary = info.PrimaryPart
    if not primary or not primary.Parent then
        unfocusCart()
        return
    end
    if primary.Position.Y < CONFIG.FellOffMapY then
        unfocusCart()
        return
    end
end

-- Seat ride
function rideSeat(seat)
    local char = getCharacter()
    if not char then return end
    local humanoid = char:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end

    if humanoid.Sit or humanoid.SeatPart then
        humanoid.Sit = false
        local t = 0
        while humanoid.SeatPart and t < 1 do
            t += task.wait(0.05)
        end
    end

    local target = seat.CFrame * CFrame.new(0, 1.8, 0)
    char:PivotTo(target)
    task.wait(0.05)
    humanoid.Sit = true
    pcall(function()
        if seat:IsA("Seat") or seat:IsA("VehicleSeat") then
            seat:Sit(humanoid)
        end
    end)
end

-- Mouse-select cart
function stopCartSelection()
    if SelectionConnection then
        cleanConnection(SelectionConnection)
        SelectionConnection = nil
    end
    SelectingCart = false
    if Gui.SelectBtn then
        Gui.SelectBtn.Text = "SELECT"
        playTween(Gui.SelectBtn, {BackgroundColor3 = Color3.fromRGB(40, 40, 40)}, 0.15)
    end
    Mouse.Icon = ""
end

function startCartSelection()
    if SelectingCart then
        stopCartSelection()
        return
    end
    SelectingCart = true
    if Gui.SelectBtn then
        Gui.SelectBtn.Text = "CLICK A CART"
        playTween(Gui.SelectBtn, {BackgroundColor3 = CONFIG.AccentColor}, 0.15)
    end
    Mouse.Icon = "rbxasset://SystemCursors/Crosshair"

    SelectionConnection = Mouse.Button1Down:Connect(function()
        local target = Mouse.Target
        if target then
            local model = getTopModel(target)
            if model and Registry[model] then
                focusCart(Registry[model])
                stopCartSelection()
            else
                -- Allow clicking anywhere else to keep trying until right-click/Esc
            end
        end
    end)
end

    end)

    end)

    return container, isRiding, riderName
end

function refreshCartList()
    local list = {}
    for _, info in pairs(Registry) do
        table.insert(list, info)
    end

    table.sort(list, function(a, b)
        local aRiding, aRider = getCartRiderInfo(a)
        local bRiding, bRider = getCartRiderInfo(b)
        if aRiding and not bRiding then return true end
        if not aRiding and bRiding then return false end
        local aKey = aRider or a.Model.Name
        local bKey = bRider or b.Model.Name
        return string.lower(aKey) < string.lower(bKey)
    end)

    clearRows(Gui.CartsPage, CartRowCache)

    local allRow = buildAllCartsRow()
    allRow.Parent = Gui.CartsPage
    table.insert(CartRowCache, allRow)

    local ridingCount = 0
    for idx, info in ipairs(list) do
        local isRiding, riderName = getCartRiderInfo(info)
        if isRiding then ridingCount += 1 end
        local row, _, _ = buildCartRow(info, idx)
        row.Parent = Gui.CartsPage
        table.insert(CartRowCache, row)
    end

    Gui.StatusLabel.Text = string.format("Carts: %d   •   Riding: %d", #list, ridingCount)
    Gui.PulseDot.BackgroundColor3 = ridingCount > 0 and CONFIG.RidingColor or CONFIG.IdleColor
end

    titleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Dragging = true
            DragStartPos = Vector2.new(input.Position.X, input.Position.Y)
            FrameStartPos = mainFrame.AbsolutePosition
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if Dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = Vector2.new(input.Position.X, input.Position.Y) - DragStartPos
            local newTopLeft = FrameStartPos + delta
            local size = mainFrame.AbsoluteSize
            local anchor = mainFrame.AnchorPoint
            mainFrame.Position = UDim2.new(0, newTopLeft.X + size.X * anchor.X, 0, newTopLeft.Y + size.Y * anchor.Y)
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Dragging = false
        end
    end)

    Gui.ScreenGui = screenGui
    Gui.MainFrame = mainFrame
    Gui.ContentFrame = contentFrame
    Gui.StatusLabel = statusLabel
    Gui.PulseDot = pulseDot
    Gui.CartsPage = cartsPage
    Gui.ControlPanel = controlPanel
    Gui.FocusName = focusName
    Gui.ViewerFrame = viewerFrame
    Gui.ViewerCamera = viewerCamera
    Gui.LockBtn = lockBtn
    Gui.StartBtn = startBtn
    Gui.UpBtn = upBtn
    Gui.DownBtn = downBtn
    Gui.SpamUpBtn = spamUpBtn
    Gui.SpamDownBtn = spamDownBtn
    Gui.MinimizeBtn = minimizeBtn
    Gui.SelectBtn = selectBtn
end

function toggleGuiExpanded()
    IsExpanded = not IsExpanded
    local mainFrame = Gui.MainFrame
    local contentFrame = Gui.ContentFrame
    local minimizeBtn = Gui.MinimizeBtn

    if IsExpanded then
        contentFrame.Visible = true
        playTween(mainFrame, {Size = UDim2.new(0, CONFIG.GuiSize.X, 0, CONFIG.GuiSize.Y)}, 0.2)
        minimizeBtn.Text = "−"
    else
        playTween(mainFrame, {Size = UDim2.new(0, CONFIG.GuiSize.X, 0, CONFIG.TitleBarHeight)}, 0.2)
        task.delay(0.2, function()
            if not IsExpanded then
                contentFrame.Visible = false
            end
        end)
        minimizeBtn.Text = "+"
    end
end

-- Button wiring
function wireControlButtons()
    local function hasFocus()
        return FocusedCart ~= nil or AllCartsMode
    end

    Gui.LockBtn.MouseButton1Click:Connect(function()
        if not hasFocus() then return end

        if AllCartsMode then
            AllCartsLockActive = not AllCartsLockActive
            for _, info in pairs(Registry) do
                if info.LockActive ~= AllCartsLockActive then
                    toggleLock(info, AllCartsLockActive)
                end
            end
        else
            toggleLock(FocusedCart)
        end
        updateLockButtonVisual()
    end)

    Gui.UpBtn.MouseButton1Click:Connect(function()
        if not hasFocus() then return end
        if AllCartsMode then
            pressButtonOnAll("Up")
        else
            guiPressButton(FocusedCart, "Up")
        end
    end)

    Gui.DownBtn.MouseButton1Click:Connect(function()
        if not hasFocus() then return end
        if AllCartsMode then
            pressButtonOnAll("Down")
        else
            guiPressButton(FocusedCart, "Down")
        end
    end)

    Gui.StartBtn.MouseButton1Click:Connect(function()
        if not hasFocus() then return end
        if AllCartsMode then
            pressButtonOnAll("Toggle")
        else
            guiPressButton(FocusedCart, "Toggle")
        end
    end)

    Gui.SpamUpBtn.MouseButton1Click:Connect(function()
        if not hasFocus() then return end
        SpamUpActive = not SpamUpActive
        if SpamUpActive then
            playTween(Gui.SpamUpBtn, {BackgroundColor3 = CONFIG.SuccessColor:Lerp(Color3.new(1,1,1), 0.3)}, 0.1)
            task.spawn(function()
                while SpamUpActive do
                    if AllCartsMode then
                        pressButtonOnAll("Up")
                    elseif FocusedCart then
                        guiPressButton(FocusedCart, "Up")
                    end
                    task.wait()
                end
            end)
        else
            playTween(Gui.SpamUpBtn, {BackgroundColor3 = CONFIG.SuccessColor}, 0.1)
        end
    end)

    Gui.SpamDownBtn.MouseButton1Click:Connect(function()
        if not hasFocus() then return end
        SpamDownActive = not SpamDownActive
        if SpamDownActive then
            playTween(Gui.SpamDownBtn, {BackgroundColor3 = CONFIG.SuccessColor:Lerp(Color3.new(1,1,1), 0.3)}, 0.1)
            task.spawn(function()
                while SpamDownActive do
                    if AllCartsMode then
                        pressButtonOnAll("Down")
                    elseif FocusedCart then
                        guiPressButton(FocusedCart, "Down")
                    end
                    task.wait()
                end
            end)
        else
            playTween(Gui.SpamDownBtn, {BackgroundColor3 = CONFIG.SuccessColor}, 0.1)
        end
    end)
end

-- Main loops
function startLoops()
    task.spawn(function()
        while true do
            scanWorkspace()
            task.wait(CONFIG.RegistryScanInterval)
        end
    end)

    task.spawn(function()
        while true do
            refreshCartList()
            for _, info in pairs(Registry) do
                if EspEnabled then
                    local isRiding, riderName = getCartRiderInfo(info)
                    if isRiding then
                        setEspRidden(info, riderName)
                    else
                        setEspIdle(info)
                    end
                end
            end
            task.wait(CONFIG.RefreshInterval)
        end
    end)

    RunService.RenderStepped:Connect(function()
        checkFocusedCart()

        if FocusedCart and FocusedCart.ViewerClone then
            local clone = FocusedCart.ViewerClone
            local angle = tick() * CONFIG.OrbitSpeed
            local extents = clone:GetExtentsSize()
            local maxDim = math.max(extents.X, extents.Y, extents.Z)
            local radius = math.max(maxDim * 1.9, 5)
            local cam = Gui.ViewerCamera
            cam.CFrame = CFrame.lookAt(Vector3.new(math.cos(angle) * radius, radius * 0.35, math.sin(angle) * radius), Vector3.new(0, 0, 0))
        end
    end)

    -- Per-frame lock guards
    RunService.Heartbeat:Connect(function()
        for _, info in pairs(Registry) do
            if info.LockActive then
                -- Toggle (Start/On) red-state guard
                local toggleCd = info.Buttons.Toggle
                if toggleCd and info.LockedToggleRed ~= nil then
                    local part = toggleCd.Parent
                    if part and part:IsA("BasePart") then
                        local nowRed = isRedColor(part.Color)
                        if info.ToggleArmed then
                            if nowRed ~= info.LockedToggleRed then
                                if tick() > (info.Guard[toggleCd] or 0) and tick() > (info.LastCounter["Toggle"] or 0) then
                                    info.LastCounter["Toggle"] = tick() + CONFIG.LockCounterDebounce
                                    guiPressButton(info, "Toggle")
                                end
                                info.ToggleArmed = false
                            end
                        else
                            if nowRed == info.LockedToggleRed then
                                info.ToggleArmed = true
                            end
                        end
                    end
                end

                -- Speed-based Up/Down guard
                if info.LockedSpeed ~= nil then
                    local current = getCartSpeed(info)
                    if current ~= nil then
                        local diff = current - info.LockedSpeed
                        if math.abs(diff) > 0.5 then
                            local upCd = info.Buttons.Up
                            local downCd = info.Buttons.Down
                            if diff < 0 and upCd and tick() > (info.Guard[upCd] or 0) and tick() > (info.LastCounter["Up"] or 0) then
                                info.LastCounter["Up"] = tick() + CONFIG.LockCounterDebounce
                                guiPressButton(info, "Up")
                            elseif diff > 0 and downCd and tick() > (info.Guard[downCd] or 0) and tick() > (info.LastCounter["Down"] or 0) then
                                info.LastCounter["Down"] = tick() + CONFIG.LockCounterDebounce
                                guiPressButton(info, "Down")
                            end
                        end
                    end
                end
            end
        end
    end)

    UserInputService.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed then return end
        if input.KeyCode == CONFIG.ToggleKey then
            IsOpen = not IsOpen
            Gui.MainFrame.Visible = IsOpen
        end
        if SelectingCart and (input.UserInputType == Enum.UserInputType.MouseButton2 or input.KeyCode == Enum.KeyCode.Escape) then
            stopCartSelection()
        end
    end)
end

-- Descendant events
table.insert(Connections, Workspace.DescendantAdded:Connect(function(instance)
    if not instance:IsA("BasePart") then return end
    task.defer(function()
        local role = getButtonName(instance)
        if role and findClickDetector(instance) then
            local model = getTopModel(instance)
            if model then
                registerCart(model)
            end
        end
        if instance:IsA("Seat") or instance:IsA("VehicleSeat") then
            local model = getTopModel(instance)
            if model and Registry[model] then
                scanModelForParts(Registry[model])
            end
        end
    end)
end))

table.insert(Connections, Workspace.DescendantRemoving:Connect(function(instance)
    local model = nil
    if instance:IsA("Model") then
        model = instance
    elseif instance:IsA("BasePart") then
        model = getTopModel(instance)
    end
    if model and Registry[model] and not isInWorkspace(model) then
        unregisterCart(model)
    end
end))

buildGui()

scanWorkspace()
refreshCartList()
startLoops()
